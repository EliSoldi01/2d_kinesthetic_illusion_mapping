 30def add_pattern_pair(df):
 31    """
 32    Clean the pattern, duration and coordinate columns and add the pattern pair.
 33
 34    The biceps and triceps patterns are converted to strings and
 35    zero-padded to 3 characters. Duration, x and y are converted to
 36    numeric values. Rows with missing duration, x, y or patterns are
 37    dropped. The new column "pattern_pair" (format "BBB_TTT") is inserted
 38    right after "pattern_triceps".
 39
 40    Parameters
 41    ----------
 42    df : pandas.DataFrame
 43        Trial-level experimental data. Must contain the columns
 44        "pattern_biceps", "pattern_triceps", "duration", "x" and "y".
 45
 46    Returns
 47    -------
 48    pandas.DataFrame
 49        Cleaned copy of df with the new column "pattern_pair".
 50    """
 51    df = df.copy()
 52
 53    df["pattern_biceps"] = df["pattern_biceps"].astype(str).str.zfill(3)
 54    df["pattern_triceps"] = df["pattern_triceps"].astype(str).str.zfill(3)
 55
 56    df["duration"] = pd.to_numeric(df["duration"], errors="coerce")
 57    df = df.dropna(subset=["duration"])
 58
 59    df["pattern_pair"] = df["pattern_biceps"] + "_" + df["pattern_triceps"]
 60
 61    df.insert(df.columns.get_loc("pattern_triceps") + 1, "pattern_pair", df.pop("pattern_pair"))
 62
 63    df["x"] = pd.to_numeric(df["x"], errors="coerce")
 64    df["y"] = pd.to_numeric(df["y"], errors="coerce")
 65
 66    df = df.dropna(subset=["x", "y", "pattern_biceps", "pattern_triceps", "pattern_pair"])
 67
 68    return df

 71def add_angle(df, protocol):
 72    """
 73    Add the angular deviation of each trial.
 74
 75    The angle is computed with geometry.compute_angle, rounded to two
 76    decimals and inserted right after the "vividness" column.
 77
 78    Required columns:
 79        x
 80        y
 81        vividness
 82        forearm_cm
 83        forearm_angle_deg
 84
 85    Parameters
 86    ----------
 87    df : pandas.DataFrame
 88        Trial-level data including the subject geometry columns.
 89
 90    protocol : dict
 91        Protocol configuration. The start cell, cell size and tested arm
 92        are read from protocol["grid"]["start_cell"],
 93        protocol["grid"]["cell_size_cm"] and protocol["arm"].
 94
 95    Returns
 96    -------
 97    pandas.DataFrame
 98        Copy of df with the new column "angle_deg" (positive = extension,
 99        negative = flexion).
100    """
101
102    df = df.copy()
103
104    start_cell = protocol["grid"]["start_cell"]
105    cell_size = protocol["grid"]["cell_size_cm"]
106    arm = protocol["arm"]
107
108    angle_deg = df.apply(
109        lambda row: geometry.compute_angle(
110            start_cell=start_cell,
111            index_cell=(row["x"], row["y"]),
112            forearm_cm=row["forearm_cm"],
113            forearm_angle_deg=row["forearm_angle_deg"],
114            cell_size=cell_size,
115            arm=arm,
116        ),
117        axis=1,
118    ).round(2)
119
120    df.insert(df.columns.get_loc("vividness") + 1, "angle_deg", angle_deg)
121
122    return df

156def add_expected_illusion(df, protocol):
157    """
158    Add the expected kinesthetic illusion for each pattern pair,
159    only if the column does not already exist.
160
161    The new column "expected_kinesthetic_illusion" is inserted right
162    after "pattern_pair". The effect of each pattern is read from the
163    protocol definitions, matching "text" with the pattern pair.
164
165    Parameters
166    ----------
167    df : pandas.DataFrame
168        Trial-level data. Must contain the column "pattern_pair".
169
170    protocol : dict
171        Protocol configuration. The pattern definitions are read from
172        protocol["patterns"]["definitions"].
173
174    Returns
175    -------
176    pandas.DataFrame
177        Copy of df with the new column, or df unchanged if the column
178        already exists.
179    """
180    column_name = "expected_kinesthetic_illusion"
181
182    # Se la colonna esiste già, restituisce il dataframe così com'è
183    if column_name in df.columns:
184        return df
185
186    df = df.copy()
187
188    effect_map = {pattern["text"]: pattern["effect"] for pattern in protocol["patterns"]["definitions"]}
189
190    expected_illusion = df["pattern_pair"].map(effect_map)
191
192    df.insert(df.columns.get_loc("pattern_pair") + 1, column_name, expected_illusion)
193
194    return df

257def prepare_data(df_main, df_subject, protocol, output_path=None):
258    """
259    Complete data preparation pipeline.
260
261    Steps:
262        1. Clean the data and add "pattern_pair".
263        2. Merge the subject data.
264        3. Add protocol metadata.
265        4. Add the expected kinesthetic illusion.
266        5. Add the angular deviation ("angle_deg").
267        6. Add the presentation order.
268        7. Optionally save the result.
269
270    Parameters
271    ----------
272    df_main : pandas.DataFrame
273        Trial-level experimental data.
274
275    df_subject : pandas.DataFrame
276        Subjects worksheet.
277
278    protocol : dict
279        Protocol configuration.
280
281    output_path : str or Path, optional
282        Directory where "data_processed.xlsx" is saved. If None, the
283        data are not saved.
284
285    Returns
286    -------
287    pandas.DataFrame
288        Prepared experimental data.
289    """
290
291    df = add_pattern_pair(df_main)
292
293    df = merge_subject_data(df, df_subject)
294
295    df = add_protocol_metadata(df, protocol)
296
297    df = add_expected_illusion(df, protocol)
298
299    df = add_angle(df, protocol)
300
301    df = add_presentation_order(df, df_subject)
302
303    if output_path is not None:
304        output_file = Path(output_path, "data_processed.xlsx")
305        save_processed_data(df, output_file)
306
307    return df

310def convert_dataframe_coordinates_to_cm(
311    df: pd.DataFrame,
312    cell_size_cm: float,
313) -> pd.DataFrame:
314    """
315    Convert the grid coordinates of a DataFrame to physical coordinates (cm).
316
317    The columns "x" and "y" are converted using
318    geometry.grid_coordinate_to_cm, so that each value corresponds to
319    the center of the cell.
320
321    Parameters
322    ----------
323    df : pandas.DataFrame
324        Data containing the columns "x" and "y" as 1-based cell
325        coordinates.
326
327    cell_size_cm : float
328        Side length of a grid cell, in cm.
329
330    Returns
331    -------
332    pandas.DataFrame
333        Copy of df with "x" and "y" expressed in cm.
334    """
335
336    df_cm = df.copy()
337
338    df_cm["x"] = df_cm["x"].apply(
339        geometry.grid_coordinate_to_cm,
340        cell_size_cm=cell_size_cm,
341    )
342    df_cm["y"] = df_cm["y"].apply(
343        geometry.grid_coordinate_to_cm,
344        cell_size_cm=cell_size_cm,
345    )
346
347    return df_cm

350def prepare_quadratic_interpolation_data(df, duration, protocol, metric="vividness"):
351    """
352    Select data for one duration and calculate the mean
353    position and mean metric for each pattern × repetition.
354
355    The grid coordinates are first converted to cm.
356
357    Parameters
358    ----------
359    df : pandas.DataFrame
360        Trial-level data containing the columns "pattern_pair", "rep",
361        "duration", "x", "y" and the selected metric.
362
363    duration : float
364        Stimulation duration to select.
365
366    protocol : dict
367        Protocol configuration. The cell size is read from
368        protocol["grid"]["cell_size_cm"].
369
370    metric : str, optional
371        Name of the column to average. Defaults to "vividness".
372
373    Returns
374    -------
375    pandas.DataFrame
376        One row per pattern × repetition, with the mean "x" and "y"
377        (in cm) and the mean of the metric.
378    """
379
380    df = convert_dataframe_coordinates_to_cm(
381        df=df,
382        cell_size_cm=protocol["grid"]["cell_size_cm"],
383    )
384
385    df_dur = df[df["duration"] == duration]
386
387    df_mean = df_dur.groupby(["pattern_pair", "rep"]).agg({"x": "mean", "y": "mean", metric: "mean"}).reset_index()
388
389    return df_mean
